async function sendMessage() { const userText = userInput.value.trim(); // أو اسم المتغير عندك if (!userText) return; // 1. اعرض رسالة المستخدم في الواجهة الأصلية appendUserMessage(userText); userInput.value = ''; try { const response = await fetch('/api/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: userText }] }] }) }); const data = await response.json(); // 2. قراءة الرد المباشر القادم من app.py let botText = ""; if (data.candidates && data.candidates[0] && data.candidates[0].content) { botText = data.candidates[0].content.parts[0].text; } else if (data.text) { botText = data.text; } else if (data.error) { botText = "خطأ من الـ API: " + JSON.stringify(data.error); } else { botText = "لم أتمكن من فهم الرد من السيرفر."; } // 3. اعرض رد الذكاء الاصطناعي في واجهتك appendBotMessage(botText); } catch (err) { console.error(err); appendBotMessage("تعذر الاتصال بالسيرفر."); } }
